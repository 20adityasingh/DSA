class Solution {
    public List<List<Integer>> permute(int[] nums) {
        List<Integer> arr = new ArrayList<>();
        List<List<Integer>> ans = new ArrayList<>();
        
        for(int num : nums){
            arr.add(num);
        }
        backTrack(arr, new ArrayList<>(), new boolean[arr.size()], ans);
        return ans;
    }
    public void backTrack(List<Integer> arr, List<Integer> temp, boolean[] used, List<List<Integer>> ans){
        if(temp.size() == arr.size()){
            ans.add(new ArrayList<>(temp));
            return;
        }

        for(int i = 0; i<arr.size(); i++){
            if(used[i]) continue;
            used[i] = true;
            temp.add(arr.get(i));

            backTrack(arr, temp, used, ans);

            used[i] = false;
            temp.remove(temp.size() - 1);
        }
    }
}
