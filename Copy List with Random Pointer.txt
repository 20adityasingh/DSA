class Solution {
    public Node copyRandomList(Node head) {


        if(head == null){
            return null;
        }

        Node curr = head;
        while(curr != null){
            Node cloneCurr = new Node(curr.val);
            cloneCurr.next = curr.next;
            curr.next = cloneCurr;
            curr = cloneCurr.next;

        }

        curr = head;
        while(curr != null){

            if(curr.random != null){

                curr.next.random = curr.random.next;

            }

            curr = curr.next.next;

        }


        curr = head;
        Node currClone = curr.next;
        Node cloneHead = curr.next;

        while(curr != null){
            curr.next = currClone.next;

            if(currClone.next != null){
                currClone.next = currClone.next.next;
            }

            curr = curr.next;
            currClone = currClone.next;

        }

        return cloneHead;
        
    }
}
