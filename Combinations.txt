class Solution {

    List<List<Integer>> comb = new ArrayList<>();

    public List<List<Integer>> combine(int n, int k) {

        List<Integer> val = new ArrayList<>();
        backtrack(n, k, val, 1);
        return comb;
        
    }

    private void backtrack(int n, int k, List<Integer> val, int idx){

        if( val.size() == k){
            comb.add(new ArrayList<>(val));
            return;
        }

        for(int i = idx; i <= n; i++){
            val.add(i);
            backtrack(n, k, val, i + 1);
            val.remove(val.size() - 1);
        }

    }
}
