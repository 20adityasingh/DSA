class Solution {
    public int numIslands(char[][] grid) {

        int count = 0;

        for(int i = 0; i< grid.length; i++){
            for(int j = 0; j< grid[0].length; j++){
                if(grid[i][j] == '1'){
                    flipTheIsland(grid, i , j);
                    count++;
                }
            }
        }

        return count;
        
    }

    public void flipTheIsland(char[][] grid, int i, int j){

        if(i >= 0 && i < grid.length && j >= 0 && j < grid[0].length && grid[i][j] == '1'){
            grid[i][j] = '0';

            flipTheIsland(grid, i-1, j);

            flipTheIsland(grid, i+1, j);

            flipTheIsland(grid, i, j-1);

            flipTheIsland(grid, i, j+1);
        }
        else{
            return;
        }
        

    }
}
