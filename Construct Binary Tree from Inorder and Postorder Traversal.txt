/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    private HashMap<Integer, Integer> inorderMap = new HashMap<>();

    private int postorderIdx;


    public TreeNode buildTree(int[] inorder, int[] postorder) {

        for(int i = 0; i < inorder.length; i++){
            inorderMap.put(inorder[i], i);
        }

        postorderIdx = postorder.length - 1;

        return helper(postorder, 0, inorder.length - 1);
        
    }

    public TreeNode helper(int[] postorder, int start, int end){

        if(start > end){
            return null;
        }


        int rootVal = postorder[postorderIdx--];

        int root_idx = inorderMap.get(rootVal);

        TreeNode root = new TreeNode(rootVal);


        root.right = helper(postorder, root_idx + 1, end);
        root.left = helper(postorder, start, root_idx - 1);

        return root;



    }
}
