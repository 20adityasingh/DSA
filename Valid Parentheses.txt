class Solution {
    public boolean isValid(String s) {
        Stack<Character> st = new Stack<>();
        for (char c : s.toCharArray()){
            if(c == '(' || c == '[' || c == '{'){
                st.push(c);
            }
            else{
                if(st.isEmpty()){
                    return false;
                }
                char e = st.pop();
                if((c == ')' && e != '(') || 
                (c == ']' && e != '[') || 
                (c == '}' && e != '{')){
                       return false; 
                }
            }
        }
        return st.isEmpty();
    }
}
