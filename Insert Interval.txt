class Solution {
    public int[][] insert(int[][] intervals, int[] newInterval) {

        ArrayList<int []> finalBoss = new ArrayList<>();

        int pos = 0;

        while (pos < intervals.length && intervals[pos][1] < newInterval[0]){
            finalBoss.add(intervals[pos]);
            pos++;
        }

        int start = newInterval[0];

        int end = newInterval[1];

        while(pos < intervals.length && intervals[pos][0] <= end){
            start = Math.min(start, intervals[pos][0]);
            end = Math.max(end, intervals[pos][1]);
            
            pos++;
        }

        finalBoss.add(new int [] {start, end});

        while(pos < intervals.length){
            finalBoss.add(intervals[pos]);
            pos++;
        }

        return finalBoss.toArray(new int[finalBoss.size()][]);

    }
}
