class Solution {
    public boolean canConstruct(String ransomNote, String magazine) {

        int magSize = magazine.length();

        int ranSize = ransomNote.length();

        if(ranSize > magSize){
            return false;
        }

        int [] freq = new int [124];

        for(int i = 0; i<ranSize; i++){
            int ch = (int) ransomNote.charAt(i);

            freq[ch]++;
        }

        int count = ranSize;

        for(int i = 0; i<magSize; i++){
            if(freq[(int)magazine.charAt(i)] > 0){
                freq[(int)magazine.charAt(i)]--;
                count--;
            }
        }
        
        if(count == 0){
            return true;
        }
        else{
            return false;
        }
    }
}
