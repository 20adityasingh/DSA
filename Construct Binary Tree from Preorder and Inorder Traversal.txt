/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    private HashMap<Integer, Integer> inorderMap = new HashMap<>();

    private int preorderRootIndex = 0;

    public TreeNode buildTree(int[] preorder, int[] inorder) {

        for(int i = 0; i < inorder.length; i++){
            inorderMap.put(inorder[i], i);
        }


        return buildTreeHelper(preorder, inorder, 0, inorder.length - 1);
        
    }

    public TreeNode buildTreeHelper(int[] preorder, int[] inorder, int start, int end){

        if(start > end) return null;

        int rootVal = preorder[preorderRootIndex];
        preorderRootIndex++;

        TreeNode root = new TreeNode(rootVal);

        int rootIdx = inorderMap.get(rootVal);


        root.left = buildTreeHelper(preorder, inorder, start, rootIdx - 1);

        root.right = buildTreeHelper(preorder, inorder, rootIdx + 1, end);

        return root;





    }
}
