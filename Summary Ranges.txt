class Solution {
    public List<String> summaryRanges(int[] nums) {
        List<String> ls = new ArrayList<>();
        
        for(int i = 0; i < nums.length; i++){
            long start = nums[i];
            
            while(i + 1 < nums.length && nums[i + 1] == nums[i] + 1){
                i++;
            }
            
            long end = nums[i];
            
            if(start == end){
                ls.add(String.valueOf(start));
            } else {
                ls.add(start + "->" + end); 
            }
        }
        
        return ls;
    }
}
